1 Introduction

In this lab, I built an IoT device using the STM32L432KC MCU, a DS1722 digital temperature sensor, and an ESP8266 Wi-Fi module. I wrote an SPI driver using the CMSIS device templates and a DS1722 driver on top of it. The MCU reads the temperature over SPI and serves a webpage (through the ESP8266 over UART) that shows the current temperature, lets the user pick the sensor resolution (8 to 12 bits), and shows and toggles the on-board LED.

1.1 Overall Design

The ESP8266 hosts a Wi-Fi access point and web server. When a browser opens http://192.168.4.1/ or clicks a button, the ESP8266 forwards the request to the MCU over UART at 125000 baud as /REQ:<tag>\n. The MCU checks the tag and updates the LED or the DS1722 resolution. It then reads the temperature from the DS1722 over SPI and sends the whole HTML page back to the ESP8266, which passes it to the browser.

  • main.c: receives requests from the ESP8266 over USART1, updates LED and resolution, reads the temperature, and sends back the webpage
  • STM32L432KC_SPI.c/.h: SPI1 master driver
  • DS1722.c/.h: DS1722 driver

1.2 SPI Driver

SPI1 is configured as master, 8-bit data, MSB first, and full duplex. CE is driven manually on PA8 as a GPIO since the DS1722’s CE is active high.

  • Baud rate: BR = 0b111 (fPCLK/256 = 80 MHz / 256 ≈ 312.5 kHz)
  • CPOL = 0, CPHA = 1 (the DS1722 latches data on the falling edge in this mode)

spiSendReceive() waits until the SPI transmit buffer has room, writes the byte to send into the data register, waits until a byte has been received, and reads it back. It then waits until the SPI bus is no longer busy, so the caller doesn’t lower CE before the last bit has finished transferring.

1.3 DS1722 Driver

Every DS1722 transaction starts with the MCU raising CE and sending an address byte over MOSI (DS1722 SDI). The address says which register to access, and its MSB selects read (0) or write (1). For a read, the DS1722 starts driving that register’s contents onto MISO (or SDO) after the 8 address clock cycles. Because SPI is full duplex, the MCU has to keep clocking to receive data, so it sends dummy 0x00 bytes on MOSI and reads what comes back on MISO.

The temperature is read in one SPI transaction which consists of the MCU sending the temperature address 0x01, then two dummy bytes. During the first dummy byte the DS1722 sends the LSB. The address then iterates to 0x02 automatically, so the MSB comes back during the second dummy byte. The two bytes form a 16-bit two’s complement value with 8 fractional bits, so the temperature in °C is the signed value divided by 256.

After the resolution changes, the driver waits the max conversion time for that resolution (75 / 150 / 300 / 600 / 1200 ms for 8–12 bits) before reading anything.

NoteSPI driver (STM32L432KC_SPI.c) 
// STM32L432KC_SPI.c
// SPI1 master driver for STM32L432KC 

#include "STM32L432KC_SPI.h"

void initSPI(int br, int cpol, int cpha) {
    // Make sure the SPI1 peripheral clock is on (APB2ENR.SPI1EN)
    RCC->APB2ENR |= RCC_APB2ENR_SPI1EN;

    // Disable SPI and start from a clean slate so re-init is safe
    SPI1->CR1 = 0;
    SPI1->CR2 = 0;

    // CR2: 8-bit data size, RXNE fires at 8 bits, Motorola frame format (FRF=0),
    // no hardware NSS output (SSOE=0), no NSS pulse (NSSP=0)
    SPI1->CR2 = _VAL2FLD(SPI_CR2_DS, 0b0111) | SPI_CR2_FRXTH;

    // CR1: baud rate, polarity, phase
    SPI1->CR1 |= _VAL2FLD(SPI_CR1_BR, br & 0x7);
    SPI1->CR1 |= _VAL2FLD(SPI_CR1_CPOL, cpol & 1);
    SPI1->CR1 |= _VAL2FLD(SPI_CR1_CPHA, cpha & 1);

    // MSB first, full duplex, 2-line (LSBFIRST, RXONLY, BIDIMODE already 0 from reset above)

    // Software slave management: internal NSS held high so master mode isn't faulted
    SPI1->CR1 |= SPI_CR1_SSM | SPI_CR1_SSI;

    // Master mode (after SSI is set)
    SPI1->CR1 |= SPI_CR1_MSTR;

    // Enable last
    SPI1->CR1 |= SPI_CR1_SPE;
}

char spiSendReceive(char send) {
    while (!(SPI1->SR & SPI_SR_TXE));                       // wait until TX buffer empty
    *(volatile uint8_t *)&SPI1->DR = (uint8_t)send;         // 8-bit write: one byte into the FIFO
    while (!(SPI1->SR & SPI_SR_RXNE));                      // wait for received byte
    uint8_t rx = *(volatile uint8_t *)&SPI1->DR;            // 8-bit read
    while (SPI1->SR & SPI_SR_BSY);                          // bus idle before caller drops CE
    return (char)rx;
}

NoteDS1722 driver (DS1722.c) 
// DS1722.c
// DS1722 digital thermometer driver over SPI (CE active high on PA8)

#include "DS1722.h"

#define DS1722_ADDR_CFG_READ   0x00
#define DS1722_ADDR_CFG_WRITE  0x80
#define DS1722_ADDR_TEMP_LSB   0x01   // MSB at 0x02; burst read auto-advances 01h -> 02h

static void ceHigh(void) { digitalWrite(DS1722_CE_PIN, 1); }
static void ceLow(void)  { digitalWrite(DS1722_CE_PIN, 0); }

// Max conversion time in ms for a given config byte (datasheet Table 5)
static uint32_t conversionMs(uint8_t cfg) {
    if (cfg & 0x08) return 1200;                 // R2=1 -> 12-bit
    switch ((cfg >> 1) & 0x3) {                  // R1:R0
        case 0:  return 75;                      // 8-bit
        case 1:  return 150;                     // 9-bit
        case 2:  return 300;                     // 10-bit
        default: return 600;                     // 11-bit
    }
}

static void writeConfig(uint8_t cfg) {
    ceHigh();
    spiSendReceive(DS1722_ADDR_CFG_WRITE);
    spiSendReceive(cfg);
    ceLow();
}

void initTempSensor(void) {
    ceLow();
    writeConfig(DS1722_CFG_12BIT);               // powers up in shutdown; clears SD
    delay_millis(TIM15, 1200);                   // let the first 12-bit conversion finish
}

void writeToTempSensor(uint8_t cfg) {
    // Accept only 111x xxx0 patterns with 1SHOT clear: 0xE0, E2, E4, E6, E8
    if ((cfg & 0xF1) != 0xE0 || ((cfg >> 1) & 0x7) > 4) return;

    writeConfig(cfg);
    delay_millis(TIM15, conversionMs(cfg));      // CE is already low while we wait
}

uint8_t readConfigFromTempSensor(void) {
    ceHigh();
    spiSendReceive(DS1722_ADDR_CFG_READ);
    uint8_t cfg = spiSendReceive(0x00);
    ceLow();
    return cfg;
}

double readFromTempSensor(void) {
    // One transaction, address 0x01, then LSB, then MSB (address auto-increments)
    ceHigh();
    spiSendReceive(DS1722_ADDR_TEMP_LSB);
    uint8_t lsb = spiSendReceive(0x00);
    uint8_t msb = spiSendReceive(0x00);
    ceLow();
    return calcTemp(msb, lsb);
}

double calcTemp(uint8_t hi, uint8_t lo) {
    // {hi, lo} is a 16-bit two's complement value with 8 fractional bits.
    // The signed cast handles negative temperatures.
    int16_t raw = (int16_t)(((uint16_t)hi << 8) | lo);
    return raw / 256.0;
}

char* webpageEnd   = "</body></html>";

// Returns 1 if des appears in request, -1 if not
int inString(char request[], char des[]) {
    if (strstr(request, des) != NULL) {return 1;}
    return -1;
}

// LED state stays between requests; only ledon/ledoff change it
int updateLEDStatus(char request[])
{
    static int led_status = 0;

    if (inString(request, "ledoff") == 1) {
        digitalWrite(LED_PIN, PIO_LOW);
        led_status = 0;
    }
    else if (inString(request, "ledon") == 1) {
        digitalWrite(LED_PIN, PIO_HIGH);
        led_status = 1;
    }

    return led_status;
}

// Returns the DS1722 config byte for a resolution button, or 0 if the request isn't one
uint8_t updateResStatus(char request[])
{
    if (inString(request, "12bit") == 1) return DS1722_CFG_12BIT;
    if (inString(request, "11bit") == 1) return DS1722_CFG_11BIT;
    if (inString(request, "10bit") == 1) return DS1722_CFG_10BIT;
    if (inString(request, "9bit")  == 1) return DS1722_CFG_9BIT;
    if (inString(request, "8bit")  == 1) return DS1722_CFG_8BIT;
    return 0;
}

// Number of resolution bits for a config byte (R2 -> 12, else 8 + R1:R0)
int resBitsFromCfg(uint8_t cfg)
{
    if (cfg & 0x08) return 12;
    return 8 + ((cfg >> 1) & 0x3);
}

// Writes value / 10^decimals as text (e.g. 235625 with decimals=4 -> "23.5625") at p.
// Handles negatives and zero-padded fractions without using printf. Returns pointer past the end.
char* fmtFixed(char *p, int value, int decimals)
{
    int scale = 1;
    for (int i = 0; i < decimals; i++) scale *= 10;

    if (value < 0) { *p++ = '-'; value = -value; }
    int whole = value / scale;
    int frac  = value % scale;

    char tmp[12];
    int n = 0;
    if (whole == 0) tmp[n++] = '0';
    while (whole > 0) { tmp[n++] = '0' + (whole % 10); whole /= 10; }
    while (n > 0) *p++ = tmp[--n];

    if (decimals > 0) {
        *p++ = '.';
        int div = scale / 10;
        for (int i = 0; i < decimals; i++) { *p++ = '0' + (frac / div) % 10; div /= 10; }
    }
    *p = '\0';
    return p;
}

void configurePins(void)
{
    pinMode(PA5, GPIO_ALT);    // SPI1_SCK
    pinMode(PA6, GPIO_ALT);    // SPI1_MISO
    pinMode(PB5, GPIO_ALT);    // SPI1_MOSI
    pinMode(DS1722_CE_PIN, GPIO_OUTPUT); // DS1722 CE (active high)
    digitalWrite(DS1722_CE_PIN, 0);      // idle low

    // Very high output speed for SCK, MISO, MOSI (2 bits per pin: PA5 -> [11:10], PA6 -> [13:12], PB5 -> [11:10])
    GPIOA->OSPEEDR |= ((3 << 10) | (3 << 12));
    GPIOB->OSPEEDR |= (3 << 10);

    // AF5 = SPI1 on PA5 ([23:20]), PA6 ([27:24]) and PB5 ([23:20]) (AFRL, 4 bits per pin)
    GPIOA->AFR[0] &= ~((0xFU << 20) | (0xFU << 24));
    GPIOB->AFR[0] &= ~(0xFU << 20);
    GPIOA->AFR[0] |=  ((5U << 20) | (5U << 24));
    GPIOB->AFR[0] |=  (5U << 20);
}

int main(void) {
  configureFlash();
  configureClock();

  gpioEnable(GPIO_PORT_A);
  gpioEnable(GPIO_PORT_B);
  gpioEnable(GPIO_PORT_C);

  pinMode(LED_PIN, GPIO_OUTPUT);

  RCC->APB2ENR |= (RCC_APB2ENR_TIM15EN);
  initTIM(TIM15);

  USART_TypeDef * USART = initUSART(USART1_ID, 125000);

  // SPI: pins first, then the peripheral (initSPI also enables the SPI1 clock)
  configurePins();
  initSPI(0b111, 0, 1);   // slowest mcu clock (PCLK/256), CPOL=0, CPHA=1 as the DS1722 requires

  initTempSensor();       // 12-bit continuous; waits for the first conversion
  int resBits = 12;



  while(1) {
    /* Wait for ESP8266 to send a request.
    Requests take the form of '/REQ:<tag>\n', with TAG begin <= 10 characters.
    */

    // Receive web request from the ESP
    char request[BUFF_LEN] = "                  "; // initialize to known value
    int charIndex = 0;

    // Keep going until you get end of line character
    while(inString(request, "\n") == -1) {
      // Wait for a complete request to be transmitted before processing
      while(!(USART->ISR & USART_ISR_RXNE));
      char c = readChar(USART);
      if (charIndex < BUFF_LEN - 1) request[charIndex++] = c;  // don't overflow the buffer
    }

    // Change resolution only if a resolution button was pressed
    uint8_t cfg = updateResStatus(request);
    if (cfg) {
      writeToTempSensor(cfg);        // waits one full conversion at the new resolution
      resBits = resBitsFromCfg(cfg);
    }

    // Read temperature and format it, no printf styff
    double temp = readFromTempSensor();
    int tenThousandths = (int)(temp * 10000.0 + (temp >= 0 ? 0.5 : -0.5));
    double tempF = temp * 9.0 / 5.0 + 32.0;
    int hundredthsF = (int)(tempF * 100.0 + (tempF >= 0 ? 0.5 : -0.5));
    char tempStatusStr[96];
    char *p = tempStatusStr;
    strcpy(p, "Temperature: ");  p += strlen(p);
    p = fmtFixed(p, tenThousandths, 4);
    strcpy(p, " C / ");          p += strlen(p);
    p = fmtFixed(p, hundredthsF, 2);
    strcpy(p, " F (");           p += strlen(p);
    p = fmtFixed(p, resBits, 0);
    strcpy(p, "-bit)");

    // Update string with current LED state
    int led_status = updateLEDStatus(request);

    char ledStatusStr[20];
    if (led_status == 1)
      sprintf(ledStatusStr, "LED is on!");
    else
      sprintf(ledStatusStr, "LED is off!");

    // transmit webpage over UART
    sendString(USART, webpageStart);

    sendString(USART, "<h2>Temperature</h2><p>");
    sendString(USART, tempStatusStr);
    sendString(USART, "</p>");
    sendString(USART, resStr);

    sendString(USART, ledStr);
    sendString(USART, "<h2>LED Status</h2><p>");
    sendString(USART, ledStatusStr);
    sendString(USART, "</p>");

    sendString(USART, webpageEnd);
  }
}

3 Results

3.1 Logic Analyzer

Channels: 1 (yellow) = SCK, 2 (green) = SDO (MCU MISO), 3 (blue) = SDI (MCU MOSI), 4 (red) = CE.

The capture shows three bytes (24 SCK cycles) inside one CE high window.

  • Byte 1: the MCU sends read address 0x01 on SDI.
  • Bytes 2 and 3: SDI sends dummy 0x00 bytes while the DS1722 drives the temperature LSB and MSB on SDO.
  • The DS1722 moves from register 0x01 to 0x02 after each byte, so one address byte is enough to get both halves of the temperature.

Byte 2 is 10100000 = 0xA0, which is 0.625 °C. Byte 3 is about 0x16 = 22 °C, the integer part. Together that is about 22.6 °C which is reasonable room temperature and matches the 22.625 °C/72.725 °F I saw on the webpage.

3.4 Resolution and Temperature Range

The DS1722 reports temperature as a 16-bit two’s complement value with 8 fractional bits (raw / 256), so the format is the same at every resolution and easily covers the -10 °C to 30 °C test range. Changing the resolution only changes how many fractional bits are valid (from 1 °C at 8-bit down to 0.0625 °C at 12-bit) and the conversion time (75 ms to 1.2 s). My code defaults to 12-bit in initTempSensor() because it gives the finest resolution, and having slower conversion doesn’t really matter since the webpage is refreshed by hand.

5.1.2 DS1722 Driver

I asked ChatGPT to “write me a C function to carry out a SPI transaction to retrieve a temperature reading from a DS1722 sensor. Make use of CMSIS libraries for the STM32L432KC.”

It gave me a driver with a 1 byte SPI1_TransferByte() helper and a DS1722_ReadTemperature() function. Its first version re-sent a register address before each data byte. It then changed to a burst read where it sent 0x01 once and then clocked in the LSB and MSB.

  • The CE polarity is wrong and the DS1722’s CE is active high, so this would never select the sensor.
  • There is no init or configuration.
  • I did the same burst read from 0x01 and the conversion (signed 16-bit value / 256).
NoteAI-generated driver as given 
#include "stm32l432xx.h"
#include <stdint.h>
#include <stdbool.h>

/* DS1722 register addresses */
#define DS1722_TEMP_LSB_REG   0x01U
#define DS1722_TEMP_MSB_REG   0x02U

/* Example chip select: PA4 */
#define DS1722_CS_LOW()  (GPIOA->BSRR = (1U << (4U + 16U)))
#define DS1722_CS_HIGH() (GPIOA->BSRR = (1U << 4U))

/*
 * Transfer one byte over SPI1.
 * SPI1 must already be initialized as an SPI master.
 */
static uint8_t SPI1_TransferByte(uint8_t data)
{
    /* Wait until transmit buffer is empty */
    while ((SPI1->SR & SPI_SR_TXE) == 0U) {
    }

    /* Send one byte */
    *(__IO uint8_t *)&SPI1->DR = data;

    /* Wait until a received byte is available */
    while ((SPI1->SR & SPI_SR_RXNE) == 0U) {
    }

    /* Return received byte */
    return *(__IO uint8_t *)&SPI1->DR;
}

/*
 * Read the DS1722 temperature.
 *
 * Returns:
 *   true  - temperature successfully read
 *   false - invalid argument
 *
 * The temperature is returned through temp_c.
 * Assumes the DS1722 is configured for 12-bit resolution.
 */
bool DS1722_ReadTemperature(float *temp_c)
{
    uint8_t temp_lsb;
    uint8_t temp_msb;
    int16_t raw;

    if (temp_c == 0)
        return false;

    /* Begin SPI transaction */
    DS1722_CS_LOW();

    /* Read LSB register: address 0x01 */
    (void)SPI1_TransferByte(DS1722_TEMP_LSB_REG);
    temp_lsb = SPI1_TransferByte(0xFFU);

    /* Read MSB register: address 0x02 */
    (void)SPI1_TransferByte(DS1722_TEMP_MSB_REG);
    temp_msb = SPI1_TransferByte(0xFFU);

    /* End SPI transaction */
    DS1722_CS_HIGH();

    /* Combine the bytes into a signed 16-bit temperature */
    raw = (int16_t)(((uint16_t)temp_msb << 8) | temp_lsb);

    /* 12-bit resolution: 1 LSB = 0.0625 degrees C */
    *temp_c = (float)raw / 256.0f;

    return true;
}
